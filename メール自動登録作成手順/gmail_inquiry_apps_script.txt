function getCategory(inquiryType) {
  const type = String(inquiryType ?? "").trim();
  const categoryMap = new Map([
    ["配送について", "配送関連"],
    ["商品の傷・不具合", "商品不良・破損"],
    ["組み立て・使い方", "組み立て・使い方"],
    ["注文の変更・キャンセル", "注文変更・キャンセル"],
    ["店舗・在庫", "店舗・在庫"],
    ["その他", "その他"],
    // 以前のテストメールにも対応
    ["配送関連", "配送関連"],
    ["商品不良・破損", "商品不良・破損"],
    ["注文変更・キャンセル", "注文変更・キャンセル"]
  ]);
  return categoryMap.get(type) || "";
}


/* ========================================
   メイン処理
======================================== */

function importGmailToSheet() {

  const lock = LockService.getScriptLock();

  if (!lock.tryLock(10000)) return;

  let ss = null;

  try {

    ss = getSpreadsheet();

    const sheet = ss.getSheetByName(
      CONFIG.SHEET_NAME
    );

    if (!sheet) {
      throw new Error("問い合わせ管理シートがありません");
    }

    validateHeaders(sheet);

    const lastRow = sheet.getLastRow();

    const registeredIds = new Set();
    const registeredNumbers = new Set();

    // 登録済みメールID・受付番号を取得
    if (lastRow > 1) {

      const rows = sheet.getRange(
        2, 1, lastRow - 1, HEADERS.length
      ).getDisplayValues();

      rows.forEach(row => {

        if (row[0]) {
          registeredNumbers.add(row[0].trim());
        }

        if (row[15]) {
          registeredIds.add(row[15]);
        }
      });
    }

    let offset = 0;
    let added = 0;
    const batchSize = 100;

    const cutoff = Date.now() -
      30 * 24 * 60 * 60 * 1000;

    while (true) {

      const threads = GmailApp.search(
        CONFIG.GMAIL_QUERY,
        offset,
        batchSize
      );

      if (threads.length === 0) break;

      for (const thread of threads) {

        for (const message of thread.getMessages()) {

          const messageId = message.getId();

          if (registeredIds.has(messageId)) continue;

          if (message.getDate().getTime() < cutoff) {
            continue;
          }

          if (!isTargetMessage(message)) continue;

          let receptionNumber = "";
          let step = "メール本文解析";

          try {

            const data = parseMailBody(
              message.getPlainBody()
            );

            receptionNumber = data["受付番号"];

            if (!receptionNumber) {
              throw new Error("受付番号が取得できません");
            }

            // 同じ受付番号が存在する場合もスキップ
            if (
              registeredNumbers.has(receptionNumber)
            ) {
              continue;
            }

            step = "受付日時変換";

            const receptionDate = data["受付日時"]
              ? parseReceptionDate(data["受付日時"])
              : "";

            step = "スプレッドシート登録";

            const category = getCategory(data["お問い合わせの種類"]);

            const row = [
              safeText(receptionNumber),             // A
              receptionDate,                         // B
              safeText(data["お客様名"]),             // C
              safeText(data["メールアドレス"]),       // D
              safeText(data["注文番号"]),             // E
              safeText(data["商品名"]),               // F
              safeText(data["問い合わせ内容（要約）"]), // G

              category,    // H カテゴリ（フォームから自動入力）
              "",          // I 緊急度
              "",          // J 対応区分
              "",          // K 担当
              "未対応",    // L ステータス
              "",          // M FAQ番号
              "",          // N 備考
              "添付確認中（残る場合は元メールを確認）", // O 添付画像
              safeText(messageId), // P メールID
              "",          // Q 対応待ち時間（数式）
              ""           // R 元メール
            ];

            const rowNumber = sheet.getLastRow() + 1;

            // 行数が不足した場合は追加
            if (rowNumber > sheet.getMaxRows()) {
              sheet.insertRowsAfter(
                sheet.getMaxRows(), 100
              );
              applyColors(sheet);
            }

            const range = sheet.getRange(
              rowNumber, 1, 1, HEADERS.length
            );

            try {

              range.setNumberFormat("@");

              sheet.getRange(rowNumber, 2)
                .setNumberFormat("MM/dd HH:mm");

              range.setValues([row]);

              // プルダウン設定
              applyDropdowns(sheet, rowNumber, 1);

              sheet.getRange(rowNumber, 7)
                .setWrap(true);

              sheet.getRange(rowNumber, 14, 1, 2)
                .setWrap(true);

              // 本文登録を先に確定する。添付保存の失敗では取り消さない。
              setWaitingTime(sheet, rowNumber);
              SpreadsheetApp.flush();

              registeredIds.add(messageId);
              registeredNumbers.add(receptionNumber);

              added++;

            } catch (writeError) {

              // 途中まで登録された内容を取り消す
              range.clearContent();
              SpreadsheetApp.flush();

              throw writeError;
            }

            // 元メールへのリンクは本文登録とは独立して処理する。
            try {
              setOriginalMailLink(sheet.getRange(rowNumber, COL.MAIL_LINK), thread);
              SpreadsheetApp.flush();
            } catch (linkError) {
              logError(ss, receptionNumber, message.getSubject(), messageId,
                "元メールリンク", linkError);
            }

            // 画像は本文登録後に保存する。失敗・非対応形式はO列とログで通知。
            try {
              const result = saveImages(message, receptionNumber);
              setImageLinks(sheet.getRange(rowNumber, COL.IMAGE), result.images, result.warnings);
              SpreadsheetApp.flush();
              result.warnings.forEach(detail => {
                logError(ss, receptionNumber, message.getSubject(), messageId,
                  "添付画像保存", new Error(detail));
              });
            } catch (imageError) {
              logError(ss, receptionNumber, message.getSubject(), messageId,
                "添付画像保存", imageError);
              try {
                sheet.getRange(rowNumber, COL.IMAGE)
                  .setValue("⚠ 添付画像の処理失敗：元メールを確認")
                  .setWrap(true);
                SpreadsheetApp.flush();
              } catch (warningError) {
                console.error("画像警告表示失敗：" + warningError.message);
              }
            }

            // 未知・空欄の種類でも問い合わせを取りこぼさない。
            // H列は空欄にして、担当者が確認できるようログへ記録する。
            if (!category) {
              try {
                logError(
                  ss, receptionNumber, message.getSubject(), messageId,
                  "カテゴリ判定",
                  new Error("お問い合わせの種類が未設定または未対応：" +
                    (data["お問い合わせの種類"] || "（空欄）"))
                );
              } catch (logErr) {
                console.error("カテゴリ判定のログ記録失敗：" + logErr.message);
              }
            }

          } catch (error) {

            logError(
              ss,
              receptionNumber,
              message.getSubject(),
              messageId,
              step,
              error
            );
          }
        }
      }

      offset += threads.length;

      if (threads.length < batchSize) break;
    }

    // 新規登録があった場合のみ自動ソート
    if (added > 0 && CONFIG.AUTO_SORT) {
      sortRows(sheet, false);
    }

    updateFilter(sheet);

    console.log("新規登録件数：" + added);

  } catch (error) {

    if (ss) {

      logError(
        ss, "", "", "",
        "システム処理",
        error
      );
    }

    throw error;

  } finally {

    lock.releaseLock();
  }
}


/* ========================================
   メール本文の解析
======================================== */

function parseMailBody(body) {
  const result = {};

  MAIL_FIELDS.forEach(field => {
    result[field.header] = "";
  });
  // 管理表に列を増やさず、カテゴリ判定にだけ使用する。
  result["お問い合わせの種類"] = "";

  const lines = body
    .replace(/\r\n?/g, "\n")
    .replace(/\u00a0/g, " ")
    .split("\n");

  let inContent = false;

  for (const rawLine of lines) {
    const line = rawLine.trim();

    // 本文中の注文番号などを別の項目として誤認識しない。
    // 現在のメール形式ではお問い合わせ内容が最後の入力項目。
    if (inContent) {
      if (line.startsWith("📎")) break;
      result["問い合わせ内容（要約）"] += "\n" + rawLine;
      continue;
    }

    if (line.startsWith("📎")) continue;

    // フォームの種類を取得（全角・半角コロン、空白に対応）。
    const typeMatch = line.match(/^お問い合わせの種類\s*[:：]\s*(.*)$/);
    if (typeMatch) {
      result["お問い合わせの種類"] = typeMatch[1].trim();
      continue;
    }

    for (const field of MAIL_FIELDS) {
      if (!line.startsWith(field.label)) continue;
      const match = line.slice(field.label.length).match(/^\s*[:：]\s*(.*)$/);
      if (!match) continue;

      result[field.header] = match[1].trim();
      if (field.header === "問い合わせ内容（要約）") {
        inContent = true;
      }
      break;
    }
  }

  Object.keys(result).forEach(key => {
    const value = result[key].trim();
    result[key] = /^(—|―|－|-|なし)$/.test(value) ? "" : value;
  });

  return result;
}


/* ========================================
   対象メールの判定
======================================== */

function isTargetMessage(message) {

  if (
    !message.getSubject()
      .startsWith(CONFIG.SUBJECT_PREFIX)
  ) {
    return false;
  }

  if (CONFIG.SENDER_EMAIL) {

    const from = message.getFrom();

    const match = from.match(/<([^<>]+)>/);

    const address = match ? match[1] : from;

    if (
      address.trim().toLowerCase() !==
      CONFIG.SENDER_EMAIL.toLowerCase()
    ) {
      return false;
    }
  }

  return true;
}


/* ========================================
   受付日時の変換
======================================== */

function parseReceptionDate(value) {

  const match = value.trim().match(
    /^(\d{4})\/(\d{1,2})\/(\d{1,2})\s+(\d{1,2}):(\d{2})$/
  );

  if (!match) {
    throw new Error("受付日時の形式が不正：" + value);
  }

  const normalized =
    match[1] + "/" +
    match[2].padStart(2, "0") + "/" +
    match[3].padStart(2, "0") + " " +
    match[4].padStart(2, "0") + ":" +
    match[5];

  const date = Utilities.parseDate(
    normalized,
    CONFIG.TIMEZONE,
    "yyyy/MM/dd HH:mm"
  );

  if (
    Utilities.formatDate(
      date,
      CONFIG.TIMEZONE,
      "yyyy/MM/dd HH:mm"
    ) !== normalized
  ) {
    throw new Error("存在しない受付日時：" + value);
  }

  return date;
}


/* ========================================
   添付画像保存（本文登録とは独立）
======================================== */

function saveImages(message, receptionNumber) {
  const images = [];
  const warnings = [];
  const attachments = message.getAttachments({
    includeInlineImages: false,
    includeAttachments: true
  });
  let folder = null;
  let folderError = null;

  attachments.forEach((attachment, index) => {
    let name = "添付ファイル " + (index + 1);
    try {
      name = attachment.getName() || name;
      const ext = (name.split(".").pop() || "").toLowerCase();
      const mime = name.includes(".") && Object.prototype.hasOwnProperty.call(IMAGE_TYPES, ext)
        ? IMAGE_TYPES[ext] : "";
      if (!mime) {
        warnings.push("非対応形式：" + name);
        return;
      }

      // JPG・PNGの代表的なファイルシグネチャを確認。
      // 他の形式を含め、画像全体の破損検査までは行わない。
      if (ext === "jpg" || ext === "jpeg" || ext === "png") {
        const signature = ext === "png"
          ? [137, 80, 78, 71, 13, 10, 26, 10]
          : [255, 216, 255];
        const bytes = attachment.getBytes();
        if (bytes.length < signature.length ||
            !signature.every((v, i) => (bytes[i] & 255) === v)) {
          throw new Error("画像形式が不正");
        }
      }

      // 画像がある場合だけDriveへアクセスする。
      // フォルダ障害でも本文登録・他のメールの処理は継続する。
      if (folderError) throw folderError;
      if (!folder) {
        try {
          folder = DriveApp.getFolderById(CONFIG.FOLDER_ID);
        } catch (error) {
          folderError = error;
          throw error;
        }
      }

      const fileName = receptionNumber + "_" + message.getId() + "_" + index + "_" + name;
      const existing = folder.getFilesByName(fileName);
      const file = existing.hasNext() ? existing.next() : folder.createFile(
        attachment.copyBlob().setName(fileName).setContentType(mime)
      );
      images.push({ name: name, url: file.getUrl() });
    } catch (error) {
      warnings.push("保存失敗：" + name + "（" + (error.message || String(error)) + "）");
    }
  });

  return { images: images, warnings: warnings };
}


/* ========================================
   画像リンク・保存失敗の表示
======================================== */

function setImageLinks(cell, images, warnings = []) {
  if (!images.length && !warnings.length) {
    cell.clearContent();
    return;
  }
  const labels = images.map(image => image.name);
  if (warnings.length) {
    labels.push("⚠ " + warnings.length + "件の添付を保存できませんでした。元メール・エラーログを確認");
  }
  const builder = SpreadsheetApp.newRichTextValue().setText(labels.join("\n"));
  let position = 0;
  images.forEach(image => {
    const end = position + image.name.length;
    builder.setLinkUrl(position, end, image.url);
    position = end + 1;
  });
  cell.setRichTextValue(builder.build()).setWrap(true);
}


/* ========================================
   対応待ち時間・元メールリンク
======================================== */

function setWaitingTime(sheet, rowNumber) {
  // B列の受付日時から経過した暦時間（休日・夜間も含む）。
  // 完了の場合は経過時間を表示しない。完了までの実績時間ではない。
  sheet.getRange(rowNumber, COL.WAIT)
    .setNumberFormat(WAIT_FORMAT)
    .setFormulaR1C1(WAIT_FORMULA);
}

function setOriginalMailLink(cell, thread) {
  const url = thread.getPermalink();
  if (!url) throw new Error("元メールのURLが取得できません");
  cell.setRichTextValue(SpreadsheetApp.newRichTextValue()
    .setText("メールを開く").setLinkUrl(url).build());
}

function writeWaitingFormulas(sheet, rows) {
  if (!rows.length) return;
  const formulas = rows.map(row => [row[0] ? WAIT_FORMULA : ""]);
  sheet.getRange(2, COL.WAIT, rows.length, 1)
    .setNumberFormat(WAIT_FORMAT).setFormulasR1C1(formulas);
}

// 更新時に既存行のQ・R列も設定する。既存のA～P列は書き換えない。
function initializeExtraColumns(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  const rows = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getDisplayValues();
  writeWaitingFormulas(sheet, rows);
  rows.forEach((row, index) => {
    const messageId = row[COL.MAIL_ID - 1];
    if (!row[0] || !messageId || row[COL.MAIL_LINK - 1]) return;
    try {
      const message = GmailApp.getMessageById(messageId);
      if (!message) throw new Error("元メールが見つかりません");
      setOriginalMailLink(sheet.getRange(index + 2, COL.MAIL_LINK), message.getThread());
    } catch (error) {
      // 削除済み・別アカウントのメールなど。既存行の更新を止めない。
      logError(sheet.getParent(), row[0], "", messageId, "元メールリンク", error);
    }
  });
}


/* ========================================
   プルダウン設定
======================================== */

function applyDropdowns(sheet, startRow, rowCount) {

  Object.entries(DROPDOWNS).forEach(([col, values]) => {

    const column = col.charCodeAt(0) - 64;

    const rule = SpreadsheetApp
      .newDataValidation()
      .requireValueInList(values, true)
      .setAllowInvalid(true)
      .build();

    sheet.getRange(
      startRow,
      column,
      rowCount,
      1
    ).setDataValidation(rule);
  });
}


/* ========================================
   緊急度・ステータスの色分け
======================================== */

function applyColors(sheet) {

  const priorityRange = sheet.getRange("I2:I");
  const statusRange = sheet.getRange("L2:L");

  const settings = [
    ["高", "#f4cccc", priorityRange],
    ["中", "#fff2cc", priorityRange],
    ["低", "#d9ead3", priorityRange],

    ["未対応", "#f4cccc", statusRange],
    ["対応中", "#fff2cc", statusRange],
    ["完了", "#d9ead3", statusRange]
  ];

  // 緊急度・ステータス以外の既存ルールを保持
  const rules = sheet.getConditionalFormatRules()
    .filter(rule =>
      !rule.getRanges().some(range =>
        range.getNumColumns() === 1 &&
        [9, 12, COL.WAIT].includes(range.getColumn())
      )
    );

  settings.forEach(([text, color, range]) => {

    rules.push(
      SpreadsheetApp.newConditionalFormatRule()
        .whenTextEqualTo(text)
        .setBackground(color)
        .setRanges([range])
        .build()
    );
  });

  const waitRange = sheet.getRange("Q2:Q");
  const warning = CONFIG.WAIT_WARNING_HOURS / 24;
  const critical = CONFIG.WAIT_CRITICAL_HOURS / 24;
  const waitRules = [
    ['=AND(ISNUMBER($Q2),$Q2>=' + critical + ')', "#f4cccc"],
    ['=AND(ISNUMBER($Q2),$Q2>=' + warning + ',$Q2<' + critical + ')', "#fff2cc"],
    ['=$Q2="対応済み"', "#d9ead3"],
    ['=$Q2="日時要確認"', "#fff2cc"]
  ];
  waitRules.forEach(([formula, color]) => {
    rules.push(SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied(formula).setBackground(color).setRanges([waitRange]).build());
  });
  sheet.setConditionalFormatRules(rules);
}


/* ========================================
   シート初期化
======================================== */

function initializeMainSheet(sheet) {

  if (sheet.getMaxColumns() < HEADERS.length) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), HEADERS.length - sheet.getMaxColumns());
  }
  if (sheet.getMaxRows() < 2) sheet.insertRowsAfter(sheet.getMaxRows(), 1);

  if (sheet.getLastRow() > 0) {

    const existing = sheet.getRange(
      1, 1, 1, HEADERS.length
    ).getDisplayValues()[0];

    // 既存のA～N列が指定の表と一致するか確認
    for (let i = 0; i < 14; i++) {

      if (existing[i] !== HEADERS[i]) {

        throw new Error(
          "既存シートの列構成が異なります：" +
          HEADERS[i]
        );
      }
    }

    // O～R列に別用途のデータがあれば上書きしない。
    for (let i = 14; i < HEADERS.length; i++) {

      if (existing[i] && existing[i] !== HEADERS[i]) {
        throw new Error("既存シートの列構成が異なります：" + HEADERS[i]);
      }

      if (!existing[i] && sheet.getLastRow() > 1) {

        const values = sheet.getRange(
          2, i + 1, sheet.getLastRow() - 1, 1
        ).getDisplayValues().flat();

        if (values.some(Boolean)) {
          throw new Error("追加先の列に既存データがあります：" + HEADERS[i]);
        }
      }
    }
  }

  sheet.getRange(1, 1, 1, HEADERS.length)
    .setValues([HEADERS])
    .setFontWeight("bold")
    .setFontColor("#ffffff")
    .setBackground("#217346")
    .setWrap(true)
    .setVerticalAlignment("middle");

  // 手入力列の見出しを青色にする
  sheet.getRange("H1:N1")
    .setBackground("#315d87");

  sheet.setFrozenRows(1);
  sheet.setFrozenColumns(2);
  sheet.setRowHeight(1, 42);

  const widths = [
    110, 115, 120, 220,
    110, 220, 400, 160,
    75, 170, 190, 100,
    100, 300, 250, 150, 145, 125
  ];

  widths.forEach((width, index) => {
    sheet.setColumnWidth(index + 1, width);
  });

  sheet.getRange("B2:B")
    .setNumberFormat("MM/dd HH:mm");

  sheet.getRange("G2:G").setWrap(true);
  sheet.getRange("N2:O").setWrap(true);

  applyDropdowns(
    sheet,
    2,
    sheet.getMaxRows() - 1
  );

  applyColors(sheet);

  // 既存のメールID列はそのまま非表示。新規のQ・R列を表示。
  sheet.hideColumns(COL.MAIL_ID);
  sheet.showColumns(COL.WAIT, 2);
  initializeExtraColumns(sheet);

  updateFilter(sheet);
}


function validateHeaders(sheet) {

  const existing = sheet.getRange(
    1, 1, 1, HEADERS.length
  ).getDisplayValues()[0];

  if (
    JSON.stringify(existing) !==
    JSON.stringify(HEADERS)
  ) {
    throw new Error("列構成が異なります。初期設定を確認してください");
  }
}


/* ========================================
   ソート機能
======================================== */

function sortRows(sheet, ascending) {

  const lastRow = sheet.getLastRow();

  if (lastRow <= 2) return;

  // 手入力部分・画像リンク・メールIDも
  // 同じ行として一緒に並べ替える
  sheet.getRange(
    2, 1, lastRow - 1, HEADERS.length
  ).sort({
    column: 2,
    ascending: ascending
  });
  // 並べ替え後も必ず同じ行のB列・L列を参照する。
  SpreadsheetApp.flush();
  writeWaitingFormulas(sheet, sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues());
}


function sortNewest() {
  manualSort(false);
}


function sortOldest() {
  manualSort(true);
}


function manualSort(ascending) {

  const lock = LockService.getScriptLock();

  lock.waitLock(10000);

  try {

    const sheet = getSpreadsheet()
      .getSheetByName(CONFIG.SHEET_NAME);

    validateHeaders(sheet);
    sortRows(sheet, ascending);

  } finally {
    lock.releaseLock();
  }
}


/* ========================================
   フィルタ機能
======================================== */

function updateFilter(sheet) {

  const filter = sheet.getFilter();

  const rowCount = sheet.getMaxRows();

  const criteria = {};

  if (filter) {

    const range = filter.getRange();

    if (
      range.getRow() === 1 &&
      range.getColumn() === 1 &&
      range.getNumRows() === rowCount &&
      range.getNumColumns() === HEADERS.length
    ) {
      return;
    }

    // 既存のフィルタ条件を保存
    if (range.getColumn() === 1) {

      for (
        let col = 1;
        col <= Math.min(range.getNumColumns(), HEADERS.length);
        col++
      ) {

        const condition =
          filter.getColumnFilterCriteria(col);

        if (condition) {
          criteria[col] = condition;
        }
      }
    }

    filter.remove();
  }

  const newFilter = sheet.getRange(
    1, 1, rowCount, HEADERS.length
  ).createFilter();

  Object.entries(criteria).forEach(([col, condition]) => {

    newFilter.setColumnFilterCriteria(
      Number(col),
      condition
    );
  });
}


/* ========================================
   エラーログ
======================================== */

function logError(
  ss,
  receptionNumber,
  subject,
  messageId,
  step,
  error
) {

  const detail = error && error.message ? error.message : String(error);
  try {
    SpreadsheetApp.flush();
    let sheet = ss.getSheetByName(
      CONFIG.ERROR_SHEET_NAME
    );

    if (!sheet) {

      sheet = ss.insertSheet(
        CONFIG.ERROR_SHEET_NAME
      );
    }

    if (sheet.getLastRow() === 0) {

      sheet.appendRow([
        "エラー発生日時",
        "受付番号",
        "メール件名",
        "メールID",
        "処理内容",
        "エラー詳細"
      ]);

      sheet.getRange("A1:F1")
        .setBackground("#a61c00")
        .setFontColor("#ffffff")
        .setFontWeight("bold");

      sheet.setFrozenRows(1);

      sheet.setColumnWidth(1, 170);
      sheet.setColumnWidth(2, 120);
      sheet.setColumnWidth(3, 280);
      sheet.setColumnWidth(4, 160);
      sheet.setColumnWidth(5, 180);
      sheet.setColumnWidth(6, 450);

      sheet.getRange("A2:A")
        .setNumberFormat("yyyy/MM/dd HH:mm:ss");

      sheet.getRange("F2:F").setWrap(true);
    }

    // 同じメールの同じエラーの重複記録を防止
    const lastRow = sheet.getLastRow();

    if (messageId && lastRow > 1) {
      const duplicate = sheet
        .getRange(2, 4, lastRow - 1, 3)
        .getDisplayValues()
        .some(row =>
          row[0] === String(messageId) &&
          row[1] === String(step) &&
          row[2] === String(detail)
        );

      if (duplicate) {
        return;
      }
    }

    sheet.appendRow([
      new Date(),
      safeText(receptionNumber),
      safeText(subject),
      safeText(messageId),
      safeText(step),
      safeText(detail)
    ]);

    SpreadsheetApp.flush();
    console.error(step + "：" + detail);
  } catch (logError) {
    console.error(step + "：" + detail);
    console.error("エラーログ記録失敗：" + (logError.message || String(logError)));
  }
}


/* ========================================
   共通処理
======================================== */

function getSpreadsheet() {

  const id = PropertiesService
    .getScriptProperties()
    .getProperty("SPREADSHEET_ID");

  if (!id) {
    throw new Error("先にsetupTriggerを実行してください");
  }

  return SpreadsheetApp.openById(id);
}


// メール本文が数式として実行されることを防止
function safeText(value) {

  const text = String(value ?? "");

  return /^\s*[=+\-@]/.test(text)
    ? "'" + text
    : text;
}


/* ========================================
   初期設定
   最初に実行する関数
======================================== */

function setupTrigger() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (!ss) {
    throw new Error("スプレッドシートから実行してください");
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    if (!(CONFIG.WAIT_WARNING_HOURS >= 0 && CONFIG.WAIT_CRITICAL_HOURS > CONFIG.WAIT_WARNING_HOURS)) {
      throw new Error("待ち時間の警告基準が不正です");
    }
    ss.setSpreadsheetTimeZone(CONFIG.TIMEZONE);
    ss.setRecalculationInterval(SpreadsheetApp.RecalculationInterval.MINUTE);

    // 保存先にアクセスできなくても本文の取り込み設定は可能。
    try {
      DriveApp.getFolderById(CONFIG.FOLDER_ID);
    } catch (error) {
      logError(ss, "", "", "", "保存先フォルダ確認", error);
    }

    let sheet = ss.getSheetByName(CONFIG.SHEET_NAME);

    if (!sheet) {
      sheet = ss.insertSheet(CONFIG.SHEET_NAME);
    }

    initializeMainSheet(sheet);

    if (!ss.getSheetByName(CONFIG.ERROR_SHEET_NAME)) {
      ss.insertSheet(CONFIG.ERROR_SHEET_NAME);
    }

    SpreadsheetApp.flush();
    PropertiesService.getScriptProperties().setProperty("SPREADSHEET_ID", ss.getId());

    // 古いトリガーを削除
    ScriptApp.getProjectTriggers()
      .filter(trigger =>
        trigger.getHandlerFunction() ===
        "importGmailToSheet"
      )
      .forEach(trigger =>
        ScriptApp.deleteTrigger(trigger)
      );

    // 5分ごとの自動実行
    ScriptApp.newTrigger("importGmailToSheet")
      .timeBased()
      .everyMinutes(CONFIG.INTERVAL)
      .create();

    console.log("初期設定完了");

  } catch (error) {

    logError(
      ss, "", "", "",
      "初期設定",
      error
    );

    throw error;
  } finally {
    lock.releaseLock();
  }
}
